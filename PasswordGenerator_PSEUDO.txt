START

FUNCTION generate_password(length, use_lowercase, use_uppercase, use_numbers, use_special):

    SET lowercase = "all lowercase letters a–z"
    SET uppercase = "all uppercase letters A–Z"
    SET digits = "all digits 0–9"
    SET special = "all special characters e.g. {}[]>.,/?"

    SET characters = empty string

        IF use_lowercase is TRUE:
            ADD lowercase characters to characters

        IF use_uppercase is TRUE:
            ADD uppercase characters to characters

        IF use_numbers is TRUE:
            ADD digit characters to characters

        F use_special is TRUE:
            ADD special characters to characters

        IF characters is empty:
            RAISE error "At least one character type must be selected"

    SET password to empty string

        REPEAT length times:
            SELECT a random character from characters
            ADD the random character to password

        RETURN password



FUNCTION main():

    DISPLAY "Password Generator"

    LOOP until valid length is entered:
        ASK user for password length
        CONVERT input to integer
        IF length is greater than 0:
            EXIT loop
        OTHERWISE:
            DISPLAY "Please enter a valid positive number."

    ASK user: include lowercase?  (True/False)
    STORE result as use_lowercase (TRUE if user entered 'y' FALSE if user entered 'n')

    ASK user: include uppercase?  (True/False)
    STORE result as use_uppercase

    ASK user: include numbers?  (True/False)
    STORE result as use_numbers

    ASK user: include special characters? (True/False)
    STORE result as use_special

    TRY:
        CALL generate_password with user selections
        DISPLAY the generated password
    CATCH error:
        DISPLAY the error message



IF this file is being run directly (NOT imported as a module):
    CALL main()

END
